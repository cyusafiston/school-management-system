import os

OUTPUT_FILE = 'project_bundle.txt'
EXCLUDE_DIRS = {'venv', '__pycache__', '.git', 'media', 'staticfiles'}
EXCLUDE_EXTENSIONS = {'.pyc', '.sqlite3'}

with open(OUTPUT_FILE, 'w', encoding='utf-8') as outfile:
  for root, dirs, files in os.walk('.'):
    # Modify dirs in-place to skip excluded directories
    dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS]

    for file in files:
      if any(file.endswith(ext) for ext in EXCLUDE_EXTENSIONS):
        continue

      file_path = os.path.join(root, file)
      outfile.write(f'--- FILE: {file_path} ---\n')
      try:
        with open(file_path, 'r', encoding='utf-8') as infile:
          outfile.write(infile.read())
      except Exception as e:
        outfile.write(f'[Error reading file: {e}]')
      outfile.write('\n\n' + '=' * 50 + '\n\n')

print(f'Project successfully bundled into {OUTPUT_FILE}!')

def validate_rwandan_phone(value):
    if not value:
        return  
    
    cleaned_value = value.strip()
    pattern = r'^(07[2389]\d{7})$'
    
    if not re.match(pattern, cleaned_value):
        raise ValidationError(
            "Invalid Rwandan phone number. Must be exactly 10 digits and start with 078, 079, 072, or 073 (e.g., 0781234567)."
        )

class StudentProfile(models.Model):
  user = models.OneToOneField(
      User, on_delete=models.CASCADE, null=True, blank=True
  )
  first_name = models.CharField(max_length=50)
  last_name = models.CharField(max_length=50)
  registration_number = models.CharField(
      max_length=30, unique=True, blank=True
  )
  date_of_birth = models.DateField()
  gender = models.CharField(
      max_length=10, choices=[('Male', 'Male'), ('Female', 'Female')]
  )
  current_class = models.ForeignKey(
      ClassStream, on_delete=models.SET_NULL, null=True
  )

  father_name = models.CharField(
      max_length=100, blank=True, null=True, default='None'
  )
  father_phone = models.CharField(
      max_length=20, blank=True, null=True, validators=[validate_rwandan_phone]
  )

  mother_name = models.CharField(
      max_length=100, blank=True, null=True, default='None'
  )
  mother_phone = models.CharField(
      max_length=20, blank=True, null=True, validators=[validate_rwandan_phone]
  )
  date_enrolled = models.DateField(auto_now_add=True)

  class Meta:
    ordering = ['id']

  @property
  def age(self):
    if not self.date_of_birth:
      return None
    today = date.today()
    return (
        today.year
        - self.date_of_birth.year
        - (
            (today.month, today.day)
            < (self.date_of_birth.month, self.date_of_birth.day)
        )
    )

  def clean(self):
    super().clean()

    if self.date_of_birth:
      if self.date_of_birth > date.today():
        raise ValidationError(
            {'date_of_birth': 'Date of birth cannot be in the future!'}
        )

      if self.age < 3 or self.age > 30:
        raise ValidationError({
            'date_of_birth': (
                'Please enter a valid school age (student must be between 3'
                ' and 30 years old).'
            )
        })

    if self.father_phone:
      validate_rwandan_phone(self.father_phone)
    if self.mother_phone:
      validate_rwandan_phone(self.mother_phone)

  def save(self, *args, **kwargs):
    if not self.registration_number:
      current_year = datetime.now().year
      prefix = f'STU/{current_year}/'

      last_student = (
          StudentProfile.objects.filter(registration_number__startswith=prefix)
          .order_by('-id')
          .first()
      )

      if last_student:
        try:
          last_seq = int(last_student.registration_number.split('/')[-1])
          new_seq = last_seq + 1
        except ValueError:
          new_seq = StudentProfile.objects.count() + 1
      else:
        new_seq = 1

      self.registration_number = f'{prefix}{new_seq:03d}'

    # Automatically create a Django User account using the exact registration number as the username
    if not self.user:
      username = self.registration_number  # Keeps format like "STU/2026/001"
      default_password = 'student123'

      user = User.objects.create_user(
          username=username,
          password=default_password,
          first_name=self.first_name,
          last_name=self.last_name,
      )
      self.user = user

    self.full_clean()
    super().save(*args, **kwargs)

    if self.current_class:
      current_term = Term.objects.filter(is_current=True).first()
      if current_term:
        StudentEnrollment.objects.get_or_create(
            student=self,
            academic_year=current_term.academic_year,
            term=current_term,
            defaults={'class_stream': self.current_class, 'status': 'Active'},
        )

  def __str__(self):
    return f'{self.first_name} {self.last_name} ({self.registration_number})'

        super().clean()
        max_cat = getattr(self.subject, 'max_cat_marks', Decimal('50.00')) or Decimal('50.00')
        max_exam = getattr(self.subject, 'max_exam_marks', Decimal('50.00')) or Decimal('50.00')

        if self.cat_marks > max_cat:
            raise ValidationError({'cat_marks': f"CAT marks cannot exceed the maximum of {max_cat} for this subject."})
        if self.exam_marks > max_exam:
            raise ValidationError({'exam_marks': f"Exam marks cannot exceed the maximum of {max_exam} for this subject."})

    def save(self, *args, **kwargs):
        max_cat = getattr(self.subject, 'max_cat_marks', Decimal('50.00')) or Decimal('50.00')
        max_exam = getattr(self.subject, 'max_exam_marks', Decimal('50.00')) or Decimal('50.00')
        
        self.max_marks = (max_cat + max_exam).quantize(Decimal('0.01'))
        self.marks_obtained = ((self.cat_marks or Decimal('0.00')) + (self.exam_marks or Decimal('0.00'))).quantize(Decimal('0.01'))
        
        if self.max_marks and self.max_marks > 0:
            raw_percentage = (self.marks_obtained / self.max_marks) * Decimal('100.00')
            self.percentage = raw_percentage.quantize(Decimal('0.01'))
        else:
            self.percentage = Decimal('0.00')
            
        if self.percentage >= Decimal('80.00'):
            self.grade_letter = 'A'
            self.remarks = 'Excellent'
        elif self.percentage >= Decimal('70.00'):
            self.grade_letter = 'B'
            self.remarks = 'Very Good'
        elif self.percentage >= Decimal('60.00'):
            self.grade_letter = 'C'
            self.remarks = 'Good'
        elif self.percentage >= Decimal('50.00'):
            self.grade_letter = 'D'
            self.remarks = 'Satisfactory'
        else:
            self.grade_letter = 'F'
            self.remarks = 'Fail'
            
        self.full_clean()
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.student} - {self.subject.name}: {self.marks_obtained}/{self.max_marks} ({self.grade_letter})"

def custom_login_view(request):
    if request.user.is_authenticated:
        logout(request)

    if request.method == 'POST':
        form = AuthenticationForm(request, data=request.POST)
        if form.is_valid():
            user = form.get_user()
            login(request, user)
            return redirect_based_on_role(request)
        else:
            print("LOGIN FORM ERRORS:", form.errors)
    else:
        form = AuthenticationForm()

    return render(request, 'core/login.html', {'form': form})


def redirect_based_on_role(request):
    user = request.user
    if not user.is_authenticated:
        return redirect('login')

    role = user.profile.role if hasattr(user, 'profile') else None
    
    if role == 'Teacher':
        return redirect('teacher_dashboard')
    elif role == 'Student':
        return redirect('student_dashboard')
    elif role in ['Administrator', 'Accountant']:
        return redirect('admin:index')
    
    if user.is_superuser or user.is_staff:
        return redirect('admin:index')
        
    logout(request)
    messages.error(request, "Your account lacks a valid role configuration.")
    return redirect('login')

    if not hasattr(request.user, 'profile') or request.user.profile.role.lower() != 'teacher':
        messages.error(request, "Access denied.")
        return redirect('/login/')

    assignment = get_object_or_404(TeacherAssignment, id=assignment_id, teacher=request.user)

    class_stream = assignment.current_class
    subject = assignment.subject
    term = assignment.term

    students = StudentProfile.objects.filter(current_class=class_stream)

    if request.method == 'POST':
        for student in students:
            cat_key = f'cat_{student.id}'
            exam_key = f'exam_{student.id}'

            if cat_key in request.POST and exam_key in request.POST:
                try:
                    cat_val = Decimal(request.POST.get(cat_key) or '0.00')
                    exam_val = Decimal(request.POST.get(exam_key) or '0.00')

                    grade_obj, created = Grade.objects.get_or_create(
                        student=student,
                        subject=subject,
                        term=term,
                        defaults={'cat_marks': cat_val, 'exam_marks': exam_val}
                    )

                    if not created:
                        grade_obj.cat_marks = cat_val
                        grade_obj.exam_marks = exam_val
                    
                    grade_obj.save() # Triggers calculations & full_clean

                except Exception as e:
                    pass

        messages.success(request, f"Grades for {class_stream.name} - {subject.name} saved successfully!")
        return redirect('teacher_dashboard')

    # Attach the specific grade object directly to each student for clean template rendering
    grades_map = {g.student_id: g for g in Grade.objects.filter(subject=subject, term=term, student__in=students)}
    for student in students:
        student.grade = grades_map.get(student.id)

    context = {
        'assignment': assignment,
        'students': students,
    }
    return render(request, 'core/input_grades.html', context)

def take_daily_attendance(request, class_id=None):
    user_profile = getattr(request.user, 'profile', None)
    if not user_profile or user_profile.role.lower() != 'teacher':
        messages.error(request, "Access denied. Only teachers can take and submit daily attendance.")
        return redirect('role_redirect')

    classes = ClassStream.objects.all()
    selected_class = None
    students = []
    already_submitted = False
    current_date = date.today()

    if class_id:
        selected_class = get_object_or_404(ClassStream, id=class_id)
        students = StudentProfile.objects.filter(current_class=selected_class)

        already_submitted = DailyClassAttendance.objects.filter(
            current_class=selected_class,
            date=current_date
        ).exists()

        if request.method == 'POST' and not already_submitted:
            valid_statuses = ['Present', 'Late', 'Absent']
            attendance_data_to_create = []
            has_error = False

            for student in students:
                status_val = request.POST.get(f'status_{student.id}')
                if status_val not in valid_statuses:
                    has_error = True
                    break
                attendance_data_to_create.append(
                    DailyClassAttendance(
                        student=student,
                        current_class=selected_class,
                        status=status_val,
                        submitted_by=request.user,
                        is_locked=True
                    )
                )

            if has_error or len(attendance_data_to_create) != students.count():
                messages.error(request, "Submission failed: Every student must have a valid attendance status assigned.")
            else:
                DailyClassAttendance.objects.bulk_create(attendance_data_to_create)
                messages.success(request, f"Attendance for {selected_class.name} submitted and locked successfully!")
                return redirect('take_daily_attendance_class', class_id=selected_class.id)

    context = {
        'classes': classes,
        'selected_class': selected_class,
        'students': students,
        'already_submitted': already_submitted,
        'today': current_date,
    }
    return render(request, 'core/take_daily_attendance.html', context)

                    pass

    if class_stream:
        notices = Notice.objects.filter(
            Q(expires_at__gte=today) | Q(expires_at__isnull=True)
        ).filter(
            Q(target_class__isnull=True) | Q(target_class=class_stream)
        )
    else:

            Q(expires_at__gte=today) | Q(expires_at__isnull=True),
            target_class__isnull=True
        )

    present_count = attendance_records.filter(status='Present').count()
    late_count = attendance_records.filter(status='Late').count()

    attendance_percentage = 0
    if total_classes > 0:
        attendance_percentage = round(((present_count + late_count) / total_classes) * 100, 1)

    if total_classes == 0:
        attendance_status = "No Records"
        status_color = "text-gray-600"
        bg_color = "bg-gray-50 text-gray-600"
        icon_class = "fa-info-circle"

        attendance_status = "Normal"
        status_color = "text-green-600"
        bg_color = "bg-green-50 text-green-600"
        icon_class = "fa-check-circle"

        attendance_status = "Warning"
        status_color = "text-yellow-600"
        bg_color = "bg-yellow-50 text-yellow-600"

@login_required
def student_attendance_view(request):
    student_profile = get_object_or_404(StudentProfile, user=request.user)
    
    try:
        year = int(request.GET.get('year', datetime.now().year))
        month = int(request.GET.get('month', datetime.now().month))
    except ValueError:
        year = datetime.now().year
        month = datetime.now().month

    subject_attendance = Attendance.objects.filter(student=student_profile).order_by('-date')
    sub_total = subject_attendance.count()
    sub_present = subject_attendance.filter(status='Present').count()
    sub_late = subject_attendance.filter(status='Late').count()
    sub_absent = subject_attendance.filter(status='Absent').count()
    sub_percentage = round(((sub_present + sub_late) / sub_total) * 100, 1) if sub_total > 0 else 0

    daily_attendance = DailyClassAttendance.objects.filter(student=student_profile).order_by('-date')
    daily_total = daily_attendance.count()
    daily_present = daily_attendance.filter(status='Present').count()
    daily_late = daily_attendance.filter(status='Late').count()
    daily_absent = daily_attendance.filter(status='Absent').count()
    daily_percentage = round(((daily_present + daily_late) / daily_total) * 100, 1) if daily_total > 0 else 0

    cal = calendar.Calendar(firstweekday=0)
    month_days = cal.monthdayscalendar(year, month)
    
    monthly_daily_records = daily_attendance.filter(date__year=year, date__month=month)
    attendance_map = {record.date.day: record.status for record in monthly_daily_records}

    prev_month = month - 1 if month > 1 else 12
    prev_year = year if month > 1 else year - 1
    next_month = month + 1 if month < 12 else 1
    next_year = year if month < 12 else year + 1

    month_name = calendar.month_name[month]

    context = {
        'student_profile': student_profile,
        'subject_attendance': subject_attendance,
        'sub_total': sub_total, 'sub_present': sub_present, 'sub_late': sub_late, 'sub_absent': sub_absent, 'sub_percentage': sub_percentage,
        'daily_attendance': daily_attendance,
        'daily_total': daily_total, 'daily_present': daily_present, 'daily_late': daily_late, 'daily_absent': daily_absent, 'daily_percentage': daily_percentage,
        'month_days': month_days,
        'attendance_map': attendance_map,
        'current_year': year,
        'current_month': month,
        'month_name': month_name,
        'prev_month': prev_month, 'prev_year': prev_year,
        'next_month': next_month, 'next_year': next_year,
    }
    return render(request, 'core/student_attendance.html', context)

@login_required
def teacher_enter_grades_view(request, assignment_id):
    if not hasattr(request.user, 'profile') or request.user.profile.role.lower() != 'teacher':
        messages.error(request, "Access denied.")
        return redirect('/login/')

    assignment = get_object_or_404(TeacherAssignment, id=assignment_id, teacher=request.user)

    class_stream = assignment.current_class
    subject = assignment.subject
    term = assignment.term

    students = StudentProfile.objects.filter(current_class=class_stream)

    if request.method == 'POST':
        for student in students:
            cat_key = f'cat_{student.id}'
            exam_key = f'exam_{student.id}'

            if cat_key in request.POST and exam_key in request.POST:
                try:
                    cat_val = Decimal(request.POST.get(cat_key) or '0.00')
                    exam_val = Decimal(request.POST.get(exam_key) or '0.00')

                    grade_obj, created = Grade.objects.get_or_create(
                        student=student,
                        subject=subject,
                        term=term,
                        defaults={'cat_marks': cat_val, 'exam_marks': exam_val}
                    )

                    if not created:
                        grade_obj.cat_marks = cat_val
                        grade_obj.exam_marks = exam_val
                    
                    grade_obj.save()

                except Exception as e:
                    pass

        messages.success(request, f"Grades for {class_stream.name} - {subject.name} saved successfully!")
        return redirect('teacher_dashboard')

    # Map grades directly to students so the template can use student.grade cleanly
    grades_map = {g.student_id: g for g in Grade.objects.filter(subject=subject, term=term, student__in=students)}
    for student in students:
        student.grade = grades_map.get(student.id)

    context = {
        'assignment': assignment,
        'students': students,
    }
    return render(request, 'core/teacher_enter_grades.html', context)

<script>
function calculateTotal(studentId) {
    const catInput = document.querySelector(`input[name="cat_${studentId}"]`);
    const examInput = document.querySelector(`input[name="exam_${studentId}"]`);
    const totalCell = document.getElementById(`total_${studentId}`);
    
    let cat = parseFloat(catInput.value) || 0;
    let exam = parseFloat(examInput.value) || 0;
    
    if (cat > 50) {
        catInput.classList.add('border-red-500', 'text-red-400');
    } else {
        catInput.classList.remove('border-red-500', 'text-red-400');
    }

    if (exam > 50) {
        examInput.classList.add('border-red-500', 'text-red-400');
    } else {
        examInput.classList.remove('border-red-500', 'text-red-400');
    }

    let total = cat + exam;
    totalCell.textContent = total.toFixed(2);
}

function fillZeros() {
    const inputs = document.querySelectorAll('input[type="number"]');
    inputs.forEach(input => {
        if (!input.value) {
            input.value = '0';
            const nameAttr = input.getAttribute('name');
            const studentId = nameAttr.split('_')[1];
            calculateTotal(studentId);
        }
    });
}

document.addEventListener('DOMContentLoaded', function() {
    {% for student in students %}
    calculateTotal('{{ student.id }}');
    {% endfor %}
});

--- FILE: .\core\templatetags\custom_filters.py ---
from django import template

register = template.Library()

@register.filter(name='get_item')
def get_item(dictionary, key):
    if dictionary and key:
        return dictionary.get(int(key)) or dictionary.get(str(key))
    return None
